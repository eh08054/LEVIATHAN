// Editor snippet (run with Tools/Unity/exec.sh while ModelCity.unity plays): renders one tile of the v9 home view
// (dist 238, target y 0.45, tilt 8, 16:9) at full resolution, by cropping the projection to the tile's rectangle.
// Params file (first line): x0 y0 w h (fractions of the full view, y down) outW outH outPng
var c = UnityEngine.Object.FindAnyObjectByType<Sector.ModelCity.ModelCity>();
var prm = System.IO.File.ReadAllText("C:/Users/eh080/AppData/Local/Temp/claude/09ed939f4dd96b9a/a769ce7c-62d0-4a9d-b3a9-a0f12f1c668f/scratchpad/tileparams.txt").Trim().Split(' ');
var inv = System.Globalization.CultureInfo.InvariantCulture;
float fx = float.Parse(prm[0], inv), fy = float.Parse(prm[1], inv), fw = float.Parse(prm[2], inv), fh = float.Parse(prm[3], inv);
int W = int.Parse(prm[4]), H = int.Parse(prm[5]); string outPng = prm[6];
var bf = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
var T = typeof(Sector.ModelCity.ModelCity);
if (T.GetField("cube", bf).GetValue(c) == null) T.GetMethod("Start", bf).Invoke(c, null);
c.hour = 16.5f; c.speed = 0; c.layer = "none";
T.GetField("dist", bf).SetValue(c, 238f); T.GetField("yaw", bf).SetValue(c, 0f); T.GetField("tilt", bf).SetValue(c, 8f);
var pr = c.PlanRect; T.GetField("target", bf).SetValue(c, new UnityEngine.Vector2(pr.width / 2, pr.height * 0.45f));
T.GetField("goal", bf).SetValue(c, null);
T.GetMethod("Update", bf).Invoke(c, null);
if (c.hud != null) c.hud.SetShown(false);
// the low outside city stays out of the shape tiles (GPT would read its boxes as buildings)
foreach (var n in new[] { "outface", "outroof" }) { var o = c.transform.Find(n); if (o != null) o.gameObject.SetActive(false); }
var cam = c.Cam;
var rt = new UnityEngine.RenderTexture(W, H, 24, UnityEngine.RenderTextureFormat.ARGB32);
rt.antiAliasing = 8;
cam.targetTexture = rt; cam.aspect = 16f / 9f;
// tile centre in the full view's NDC (y up), and its size; scale that window up to fill the screen
float cx = (fx + fw / 2) * 2 - 1, cy = 1 - (fy + fh / 2) * 2;
var crop = UnityEngine.Matrix4x4.identity;
crop.m00 = 1 / fw; crop.m03 = -cx / fw; crop.m11 = 1 / fh; crop.m13 = -cy / fh;
var full = cam.projectionMatrix;
var shift = cam.lensShift;
cam.projectionMatrix = crop * full;
cam.lensShift = UnityEngine.Vector2.zero;   // the shift is already in `full`; URP would apply it again
cam.Render();
cam.lensShift = shift;
UnityEngine.RenderTexture.active = rt;
var tex = new UnityEngine.Texture2D(W, H, UnityEngine.TextureFormat.RGB24, false);
tex.ReadPixels(new UnityEngine.Rect(0, 0, W, H), 0, 0); tex.Apply();
System.IO.File.WriteAllBytes(outPng, tex.EncodeToPNG());
cam.ResetProjectionMatrix(); cam.targetTexture = null; cam.ResetAspect(); UnityEngine.RenderTexture.active = null; rt.Release();
return "ok " + outPng;
